spring.application.name=url-shortener

# In-memory database for the prototype; production would use PostgreSQL.
spring.datasource.url=jdbc:h2:mem:shortener;DB_CLOSE_DELAY=-1
spring.jpa.hibernate.ddl-auto=update
spring.jpa.open-in-view=false

# Errors as RFC 7807 problem details
spring.mvc.problemdetails.enabled=true

# Only the health endpoint is exposed for monitoring
management.endpoints.web.exposure.include=health,info

# Redirect cache
shortener.cache.ttl-seconds=60
shortener.cache.max-size=10000

# Per-IP rate limit: burst size and sustained requests per second
shortener.ratelimit.capacity=100
shortener.ratelimit.refill-per-second=10
